    function loadUserPicksFromData() {
      if (!currentUser) return;
      const voterKey = getSafeVoterKey(currentUser.uid);
      const existing = allVotesData[voterKey] || {};
      userPicks = {};
      MEMBERS.forEach((m) => {
        userPicks[m.id] = Number((existing.picks && existing.picks[m.id]) || 0);
      });
      renderMembers();
    }

    function getUsedVotes() {
      return Object.values(userPicks).reduce((sum, v) => sum + Number(v || 0), 0);
    }

    function changeMemberVote(memberId, delta) {
      const currentVal = Number(userPicks[memberId] || 0);
      const used = getUsedVotes();
      const nextVal = currentVal + delta;

      if (nextVal < 0) return;
      if (delta > 0 && used + delta > MAX_VOTES) {
        const msg = document.getElementById("voteStatus");
        msg.className = "status-msg error-msg";
        msg.textContent = `En fazla ${MAX_VOTES} oy verebilirsiniz.`;
        return;
      }

      userPicks[memberId] = nextVal;
      document.getElementById("voteStatus").textContent = "";
      renderMembers();
    }

    async function syncPublicTotalsToFirebase(latestVotesMap) {
      const totals = {};
      MEMBERS.forEach((m) => { totals[m.id] = 0; });

      let grandTotal = 0;
      let voterCount = 0;

      Object.values(latestVotesMap || {}).forEach((entry) => {
        if (!entry || !entry.picks) return;
        voterCount += 1;
        MEMBERS.forEach((m) => {
          const v = Number(entry.picks[m.id] || 0);
          totals[m.id] += v;
          grandTotal += v;
        });
      });

      await db.ref(PUBLIC_SUMMARY_PATH).set({
        counts: totals,
        grandTotal,
        voterCount,
        updatedAt: new Date().toISOString()
      });
    }

    async function saveVotesToFirebase() {
      const msg = document.getElementById("voteStatus");
      if (!currentUser) {
        msg.className = "status-msg error-msg";
        msg.textContent = "Önce Google ile giriş yapmalısınız.";
        return;
      }

      const used = getUsedVotes();
      if (used > MAX_VOTES) {
        msg.className = "status-msg error-msg";
        msg.textContent = `Toplam oy ${MAX_VOTES} sınırını aşamaz.`;
        return;
      }

      const voterKey = getSafeVoterKey(currentUser.uid);
      const devId = getDeviceId();
      const email = (currentUser.email || "").toLowerCase();
      lockDeviceToGmail(email);

      const picksByName = {};
      MEMBERS.forEach((m) => {
        const v = Number(userPicks[m.id] || 0);
        if (v > 0) picksByName[m.name] = v;
      });

      const payload = {
        uid: currentUser.uid,
        email: currentUser.email || "",
        name: currentUser.displayName || "",
        deviceId: devId,
        picks: userPicks,
        picksByName,
        totalVotes: used,
        updatedAt: new Date().toISOString()
      };

      try {
        await db.ref(`${DB_PATH}/${voterKey}`).set(payload);

        const nextAll = { ...allVotesData, [voterKey]: payload };
        allVotesData = nextAll;
        await syncPublicTotalsToFirebase(nextAll);

        msg.className = "status-msg";
        msg.textContent = `Oylarınız kaydedildi (${used}/${MAX_VOTES} oy kullanıldı).`;
      } catch (err) {
        msg.className = "status-msg error-msg";
        msg.textContent = "Kayıt hatası: " + err.message;
      }
    }

    function renderResults() {
      const listEl = document.getElementById("resultsList");
      const metaEl = document.getElementById("resultsSummaryMeta");
      if (!listEl || !metaEl) return;

      const totals = {};
      MEMBERS.forEach((m) => { totals[m.id] = 0; });

      let grandTotal = 0;
      let voterCount = 0;

      if (publicTotalsData && publicTotalsData.counts) {
        MEMBERS.forEach((m) => {
          totals[m.id] = Number(publicTotalsData.counts[m.id] || 0);
        });
        grandTotal = Number(publicTotalsData.grandTotal || 0);
        voterCount = Number(publicTotalsData.voterCount || 0);
      } else {
        Object.values(allVotesData).forEach((entry) => {
          if (!entry || !entry.picks) return;
          voterCount += 1;
          MEMBERS.forEach((m) => {
            const v = Number(entry.picks[m.id] || 0);
            totals[m.id] += v;
            grandTotal += v;
          });
        });
      }

      const maxVotesReceived = Math.max(0, ...MEMBERS.map((m) => totals[m.id] || 0));
      metaEl.innerHTML = `Oylanacak Kişi: <strong>${MEMBERS.length}</strong> • Katılımcı Sayısı: <strong>${voterCount}</strong> • Toplam Kullanılan Oy: <strong>${grandTotal}</strong>`;

      const ranked = [...MEMBERS]
        .map((m) => ({ ...m, votes: totals[m.id] || 0 }))
        .sort((a, b) => b.votes - a.votes || a.order - b.order);

      listEl.innerHTML = "";
      ranked.forEach((m) => {
        const pct = grandTotal > 0 ? ((m.votes / grandTotal) * 100).toFixed(1) : "0.0";
        const barWidth = maxVotesReceived > 0 ? Math.round((m.votes / maxVotesReceived) * 100) : 0;
        const div = document.createElement("div");
        div.className = "summary-item";
        div.innerHTML = `
          <div>
            <span class="member-name">${m.name}</span>
            <span class="rank-tag">${m.rank}</span>
            <div class="member-sub">#${m.order} • CP: ${m.cp}</div>
          </div>
          <div class="bar-wrap"><div class="bar-fill" style="width:${barWidth}%;"></div></div>
          <div style="text-align:right; min-width:95px;">
            <strong>${m.votes} oy</strong>
            <span class="muted" style="margin-left:6px;">(%${pct})</span>
          </div>
        `;
        listEl.appendChild(div);
      });
    }

    function renderAudit() {
      const auditEl = document.getElementById("auditList");
      if (!auditEl) return;

      if (!isAdmin) {
        auditEl.innerHTML = `<p class="error-msg">Bu alan yalnızca ${ADMIN_EMAIL} hesabına açıktır.</p>`;
        return;
      }

      const q = (document.getElementById("auditSearch").value || "").trim().toLowerCase();
      const memberMap = {};
      MEMBERS.forEach((m) => { memberMap[m.id] = `${m.name} (${m.rank})`; });

      const entries = Object.values(allVotesData).filter((e) => e && e.picks);

      const filtered = entries.filter((entry) => {
        const email = (entry.email || "").toLowerCase();
        const dev = (entry.deviceId || "").toLowerCase();
        const votedNames = Object.entries(entry.picks || {})
          .filter(([, count]) => Number(count) > 0)
          .map(([id, count]) => `${(memberMap[id] || id).toLowerCase()} ${count}`)
          .join(" ");
        return !q || email.includes(q) || dev.includes(q) || votedNames.includes(q);
      });

      auditEl.innerHTML = "";
      if (filtered.length === 0) {
        auditEl.innerHTML = `<p class="muted">Henüz kayıtlı oy bulunmuyor veya arama kriteriyle eşleşmedi.</p>`;
        return;
      }

      filtered.forEach((entry) => {
        const given = Object.entries(entry.picks || {})
          .filter(([, count]) => Number(count) > 0)
          .sort((a, b) => Number(b[1]) - Number(a[1]));

        const pillsHtml = given.length
          ? given.map(([id, count]) => `<span class="pill">${memberMap[id] || id}: <strong>${count} oy</strong></span>`).join("")
          : `<span class="muted">Henüz oy dağıtımı yapılmamış.</span>`;

        const card = document.createElement("div");
        card.className = "audit-card";
        card.innerHTML = `
          <div class="audit-header">
            <div>
              <strong>${entry.email || "Bilinmeyen Hesap"}</strong>
              <div class="muted">Cihaz: ${entry.deviceId || "-"}</div>
            </div>
            <div style="text-align:right;">
              <strong>${Number(entry.totalVotes || 0)} / ${MAX_VOTES} oy</strong>
              <div class="muted">${entry.updatedAt ? new Date(entry.updatedAt).toLocaleString("tr-TR") : ""}</div>
            </div>
          </div>
          <div class="pill-list">${pillsHtml}</div>
        `;
        auditEl.appendChild(card);
      });
    }
